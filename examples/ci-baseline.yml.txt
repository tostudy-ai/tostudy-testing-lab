        with:
          node-version-file: .node-version
      - run: pnpm install --frozen-lockfile
      - run: pnpm test:baseline
      - run: pnpm test:baseline # Same source, same question, no new evidence.
      - run: pnpm exec playwright install --with-deps chromium
